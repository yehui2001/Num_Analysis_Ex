syms x1 x2 x3
f1(x1,x2,x3) = 5*x1-x2^2+x1*x3-2;
f2(x1,x2,x3) = x1*x3+4*x2-x3;
f3(x1,x2,x3) = x1^2+x2-3*x3-1;
f(x1,x2,x3) = [f1 f2 f3];
jac(x1,x2,x3) = jacobian([f1 f2 f3],[x1 x2 x3]);
p=[1 0 0];
F0 = [0 0 0];
for i=1:20
    dq = eval(F0-formula(f(p(1),p(2),p(3))));
    if norm(dq) < 1e-5
        fprintf('求解成功\t迭代次数：%d\n',i);
        p
        break;
    end
    dp = eval( formula(jac(p(1),p(2),p(3)))\dq' );
    p = p+dp';
    fprintf('\n迭代次数:%d',i);
    vpa(p,10)
end
