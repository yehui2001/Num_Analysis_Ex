Find the numerical solution of the following nonlinear equations using iteration method and Newton method respectively and show your analysis of error by using norms of vector and matrix and demonstrate your computer code：
\begin{equation}

\end{array}\right.
\end{equation}

%--------------------------------------------------
\begin{frame}[fragile]
\frametitle{Problem 2(1) Fixed-Point Python Code}
%\rule{\textwidth}{1pt}
\begin{block}{Python}
\begin{lstlisting}[language={python},
                   numberstyle=\tiny\menlo,
                   basicstyle=\small\menlo]
x = np.zeros([max_iteration, 2], dtype='float64')
err = np.zeros(max_iteration - 1, dtype='float64')
distance = 1;idx = 1;x[0, 0] = 1;x[0, 1] = 1
while(distance > acc):
    if(idx < max_iteration):
        x[idx, 0] = ((x[idx - 1, 0])**2 + (x[idx - 1, 1])**2 + 8) / 10
        x[idx, 1] = (x[idx - 1, 0] + (x[idx - 1, 0] * x[idx - 1, 1])**2 + 7) / 12
        distance = (x[idx - 1, 0] - x[idx, 0])**2 + (x[idx - 1, 1] - x[idx, 1])**2
        err[idx - 1] = distance
        idx = idx + 1 
    else:
        break  
\end{lstlisting}
\end{block}

\begin{lstlisting}[language={matlab},

syms x1 x2;
f1(x1,x2) = x1^2 - 10*x1 + x2^2 + 8;
f2(x1,x2) = x1*x2^2 + x1 - 12*x2 + 7;
F(x1,x2) = [f1;f2];x_solve(:,1) = [1;1];
J(x1,x2) = jacobian(F,[x1 x2]);
dx_Nr =[0;0];
for i = 1:N
    dx_Nr(:,i) = J(x_solve(1,i),x_solve(2,i))\(-F(x_solve(1,i),x_solve(2,i)));
    if(sqrt(sum(dx_Nr(:,i).^2)) < epsilon)
        x_solve(:,i+1) = x_solve(:,i) + dx_Nr(:,i);
        break;
    else
        x_solve(:,i+1) = x_solve(:,i) + dx_Nr(:,i);
    end end
\end{lstlisting}
